cmake_minimum_required(VERSION 3.16)
project(FloksC VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(FrnlC 0.2 CONFIG REQUIRED)
add_library(floksc STATIC src/floksc.c src/floksc_frnl.c src/serialization.c src/layout.c)
add_library(FloksC::floksc ALIAS floksc)
target_compile_features(floksc PUBLIC c_std_11)
set_target_properties(floksc PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(floksc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(floksc PUBLIC FrnlC::frnl_semantic)
if(UNIX)
 target_link_libraries(floksc PUBLIC m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(floksc PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(floksc_native tests/native.c)
 target_link_libraries(floksc_native PRIVATE floksc)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(floksc_native PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 endif()
 add_test(NAME floksc_native COMMAND floksc_native)
 add_executable(floksc_serialization tests/serialization.c)
 target_link_libraries(floksc_serialization PRIVATE floksc)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(floksc_serialization PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 endif()
 add_test(NAME floksc_serialization COMMAND floksc_serialization)
 add_executable(floksc_layout tests/layout.c)
 target_link_libraries(floksc_layout PRIVATE floksc)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(floksc_layout PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 endif()
 add_test(NAME floksc_layout COMMAND floksc_layout)
endif()
install(TARGETS floksc EXPORT FloksCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FloksCTargets NAMESPACE FloksC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksC)
configure_package_config_file(cmake/FloksCConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/FloksCConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksC)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/FloksCConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/FloksCConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/FloksCConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksC)
