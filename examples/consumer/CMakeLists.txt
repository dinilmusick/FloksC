cmake_minimum_required(VERSION 3.16)
project(FloksCExternal C)
find_package(FrnlC 0.2 CONFIG REQUIRED)
find_package(FloksC 0.1 CONFIG REQUIRED)
add_executable(floksc_external main.c)
target_link_libraries(floksc_external PRIVATE FloksC::floksc FrnlC::frnl_semantic)
set_target_properties(floksc_external PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(floksc_external PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND floksc_external)
